───────────────────────────────────────────────────────────────────────────── # CORE ENGINE # ───────────────────────────────────────────────────────────────────────────── def collect_files(input_path: str, extensions: set) -> list[Path]: """Return all matching files; handles file OR folder (recursive).""" p = Path(input_path) if p.is_file(): return [p] if p.suffix.lower() in extensions else [] files = [] for f in p.rglob("*"): if f.is_file() and f.suffix.lower() in extensions: files.append(f) return sorted(files) def safe_output_path(src: Path, out_dir: Path | None, new_suffix: str | None = None, suffix_tag: str = "_out") -> Path: """Build output path, never overwriting originals unless out_dir is same.""" stem = src.stem + (suffix_tag if out_dir is None else "") ext = new_suffix if new_suffix else src.suffix if out_dir: out_dir.mkdir(parents=True, exist_ok=True) return out_dir / (stem + ext) return src.parent / (stem + ext) # PNG/JPG → SVG ───────────────────────────────────────────────────────── def process_raster_to_svg(input_path, output_folder=None, colormode="color", log=print): if not VTRACER_OK: log("✗ vtracer not installed. Strategy: parse SVG viewBox/width/height, render shapes via Pillow's ImageDraw for basic SVGs, OR fall back to a blank canvas with a label for complex SVGs. For production-quality tracing use Inkscape CLI if installed (detected automatically). """ import subprocess, shutil, tempfile # ── Try Inkscape CLI first (best quality, optional) ────────────────── inkscape = shutil.which("inkscape") if inkscape: scale = dpi / 96.0 with tempfile.NamedTemporaryFile(suffix=".png", delete=False) as tmp: tmp_path = tmp.name subprocess.run( [inkscape, str(svg_path), f"--export-filename={tmp_path}", f"--export-dpi={dpi}"], check=True, capture_output=True ) img = Image.open(tmp_path).copy() Path(tmp_path).unlink(missing_ok=True) return img # ── Pure-Python fallback: parse SVG dimensions → white canvas ──────── import xml.etree.ElementTree as ET tree = ET.parse(str(svg_path)) root = tree.getroot() ns = {"svg": "http://www.w3.org/2000/svg"} def _dim(val, default=512): if val is None: return default val = val.strip().rstrip("pt").rstrip("px").rstrip("mm").strip() try: return max(1, int(float(val) * (dpi / 96.0))) except ValueError: return default vb = root.get("viewBox", "") if vb: parts = vb.replace(",", " ").split() w = _dim(parts[2]) if len(parts) >= 3 else _dim(root.get("width")) h = _dim(parts[3]) if len(parts) >= 4 else _dim(root.get("height")) else: w = _dim(root.get("width")) h = _dim(root.get("height")) # Render a white canvas — basic but guaranteed to work img = Image.new("RGBA", (w, h), (255, 255, 255, 255)) draw = ImageDraw.Draw(img) draw.text((10, 10), f"SVG: {svg_path.name}\n(Install Inkscape for full render)", fill=(180, 180, 180)) return img # ── 4. Watermark ───────────────────────────────────────────────────────────── def process_watermark(input_path, output_folder=None, wm_text="© Your Brand", opacity=100, position="bottom-right", log=print): if not PIL_OK: log("✗ Pillow not installed."); return files = collect_files(input_path, IMG_EXTS) if not files: log("✗ No images found."); return out_base = Path(output_folder) if output_folder else None ok, fail = 0, 0 for f in files: try: rel = f.relative_to(Path(input_path).parent if Path(input_path).is_file() else input_path) odir = (out_base / rel.parent) if out_base else None opath = safe_output_path(f, odir, suffix_tag="_wm") img = Image.open(f).convert("RGBA") txt_layer = Image.new("RGBA", img.size, (255,255,255,0)) draw = ImageDraw.Draw(txt_layer) try: font = ImageFont.truetype("arial.ttf", max(16, img.width // 40)) except Exception: font = ImageFont.load_default() bbox = draw.textbbox((0,0), wm_text, font=font) tw, th = bbox[2]-bbox[0], bbox[3]-bbox[1] pad = 20 pos_map = { "top-left": (pad, pad), "top-right": (img.width - tw - pad, pad), "bottom-left": (pad, img.height - th - pad), "bottom-right": (img.width - tw - pad, img.height - th - pad), "center": ((img.width-tw)//2, (img.height-th)//2), } xy = pos_map.get(position, pos_map["bottom-right"]) draw.text(xy, wm_text, fill=(255,255,255,opacity), font=font) draw.text((xy[0]+2, xy[1]+2), wm_text, fill=(0,0,0,opacity//2), font=font) out = Image.alpha_composite(img, txt_layer).convert("RGB") out.save(opath) log(f"✓ {f.name} → {opath.name}") ok += 1 except Exception as e: log(f"✗ {f.name}: {e}"); fail += 1 log(f"\n── Done: {ok} watermarked, {fail} failed ──") # ── 7.
